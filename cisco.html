Default CLI commands:

  1. enable
  2. configure terminal
  3. interface Vlan1
  4. ip address 192.168.1.125 255.255.255.128
  5. no shutdown
  6. exit
  7. ip default-gateway 192.168.1.126
  8. copy running-config startup-config

  1. R1(config)# service dhcp
  2. R1# configure terminal
  3. R1(config)# ip dhcp excluded-address 192.168.1.1 192.168.1.10 (x.1 to x.10 is the range)
  4. R1(config)# ip dhcp pool LAN_POOL
  5. R1(config-dhcp)# network 192.168.1.0 255.255.255.128
  6. R1(config-dhcp)# default-router 192.168.1.126
  7. R1(config-dhcp)# dns-server 8.8.4.4
  8. R1(config-dhcp)# exit
  9. R1(config)# exit
  10. copy running-config startup-config
